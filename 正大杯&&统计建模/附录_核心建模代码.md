# 附录：核心建模与数据分析代码

本附录收录了项目研究过程中使用的关键 Python 代码，按照建模逻辑顺序整理。

**建模技术路线：**
1.  **数据质量检验**：使用 Cronbach's Alpha 和 EFA (探索性因子分析) 验证问卷信效度。
2.  **效能指数测算**：采用单纯熵值法 (Entropy Weight) 确定客观权重，结合模糊综合评价思想计算“上海市民健身驿站服务效能指数 (SEI)”。
3.  **驱动机制挖掘**：构建 XGBoost 非线性回归模型，并通过 SHAP (Shapley Additive Explanations) 进行模型解构。
4.  **策略矩阵构建**：结合 SHAP 重要性与现状满意度，绘制 IPA (Importance-Performance Analysis) 决策矩阵。

## 模块二：效能指数 SEI 测算 (Index Calculation)

本模块利用**熵值法**客观赋权，计算每位受访者的“服务效能指数 (SEI)”，该指数将作为后续机器学习模型的目标变量 $Y$。

### 2.1 熵值法计算权重
**代码文件**：`analysis/calc_entropy_weights.py`

```python
def entropy_weight(data):
    """
    熵值法计算权重
    :param data: DataFrame, 数值型矩阵 (行:样本, 列:指标)
    :return: Series, 各指标权重
    """
    # 1. 预处理：删除空值
    df = data.dropna().copy()
    n, m = df.shape
    
    # 2. 数据标准化 (Min-Max)
    # 为了避免 log(0)，我们将范围映射到 [0.002, 1] 
    # 原公式: (x - min) / (max - min) -> [0, 1]
    # 修正: 0.998 * (x - min) / (max - min) + 0.002
    min_val = df.min()
    max_val = df.max()
    normalized_df = 0.998 * (df - min_val) / (max_val - min_val) + 0.002
    
    # 3. 计算比重 P_ij
    # P_ij = x_ij / sum(x_ij)
    sum_col = normalized_df.sum(axis=0)
    p_matrix = normalized_df / sum_col
    
    # 4. 计算计算熵值 e_j
    # k = 1 / ln(n)
    k = 1 / np.log(n)
    e_j = -k * (p_matrix * np.log(p_matrix)).sum(axis=0)
    
    # 5. 计算差异系数 g_j
    g_j = 1 - e_j
    
    # 6. 计算权重 w_j
    w_j = g_j / g_j.sum()
    
    return w_j

if __name__ == "__main__":
    # 读取数据
    data_path = r'questionnaire_simulation\simulated_data_final_users_only.csv'
    df_user = pd.read_csv(data_path)

    # 定义评价指标列 (与 EFA 阶段一致)
    col_list = [
        'Q13a_Distance', 'Q13b_Traffic', 'Q14_FacilityType', 'Q15_Maintenance', 
        'Q16_Environment', 'Q19_Amenities', 'Q16_Smart_Device', 'Q16_Smart_Data', 
        'Q17_Hours', 'Q18_Cost', 'Q20_Staff', 'Q21_Info'
    ]
    
    # 提取目标数据
    df_calc = df_user[col_list]
    
    # 计算权重
    weights = entropy_weight(df_calc)
    
    # 格式化输出
    print("\n=== 熵值法指标权重结果 ===")
    weights_df = weights.to_frame(name='Weight')
    weights_df['Weight'] = weights_df['Weight'].round(4)
    weights_df = weights_df.sort_values(by='Weight', ascending=False)
    
    print(weights_df)
    
    # 保存结果
    save_path = r'questionnaire_simulation\analysis\weights_result.csv'
    weights_df.to_csv(save_path, encoding='utf-8-sig')
    print(f"\n权重结果已保存至: {save_path}")
```

### 2.2 计算 SEI 指数 (Fuzzy Comprehensive Evaluation Logic)
**代码文件**：`analysis/calc_fce_target.py`

```python
# 1. 读取数据和权重
data_path = r'questionnaire_simulation\simulated_data_final_users_only.csv'
weights_path = r'questionnaire_simulation\analysis\weights_result.csv'

df_user = pd.read_csv(data_path)
weights_df = pd.read_csv(weights_path, index_col=0) # 第一列是指标名

# 确保列名列表顺序一致
col_list = weights_df.index.tolist()
# 提取权重向量
W = weights_df['Weight'].values

print("指标列:", col_list)
print("权重向量:", W)

# 任务一：计算“上海市整体服务效能”得分
# 提取评分矩阵 R (Mean per factor)
means = df_user[col_list].mean()
print("\n各指标平均分:")
print(means)

# 综合得分 (1-7分制) = Sum(Mean_i * Weight_i)
# 类似于 FCE 中 B = W * R，这里 R 是均值向量
overall_score_7 = np.dot(means.values, W)

# 转换为百分制
# 公式：(Score - Min) / (Max - Min) * 100，这里 Min=1, Max=7
overall_score_100 = (overall_score_7 - 1) * (100 / 6)

print(f"\n==========================================")
print(f"上海市健身驿站整体服务效能得分为：{overall_score_100:.2f} 分")
print(f"==========================================")


# 任务二：计算每个受访者的效能指数 (SEI)
# 对每一行进行加权求和: SEI_i = Sum(Row_i * W)
# 计算 raw score (1-7)
df_user['SEI_Raw'] = df_user[col_list].dot(W)

# 归一化到 0-100
df_user['SEI_Score'] = (df_user['SEI_Raw'] - 1) * (100 / 6)

print("\n受访者效能指数 (SEI)预览:")
print(df_user[['SEI_Raw', 'SEI_Score']].head())


# 任务三：保存建模数据
# 定义需要保留的列
demographic_cols = ['Q1_Gender', 'Q2_Age', 'Q3_Job', 'Q4_District', 'Q5_Income']
score_cols = col_list
target_cols = ['SEI_Score', 'Q24_Satisfaction', 'WalkTime_Real']

# 检查是否所有列都在 df_user 中 (防止列名拼写错误)
cols_to_save = demographic_cols + score_cols + target_cols
valid_cols = [c for c in cols_to_save if c in df_user.columns]

model_data = df_user[valid_cols].copy()

# 再次检查是否有空值 (虽然之前处理过，但安全起见)
model_data = model_data.dropna()

save_path = r'questionnaire_simulation\model_ready_data.csv'
model_data.to_csv(save_path, index=False, encoding='utf-8-sig')

print(f"\n建模数据已保存至: {save_path}")
print(f"数据量: {model_data.shape}")
print(f"包含特征: {list(model_data.columns)}")

```

# 读取数据
data_path = 'questionnaire_simulation/model_ready_data.csv'
df = pd.read_csv(data_path)

# 定义指标列表 (顺序必须固定)
